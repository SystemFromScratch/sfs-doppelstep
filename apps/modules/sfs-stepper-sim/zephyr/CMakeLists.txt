# Builds the plant model of sfs-stepper-sim (sil/c/sstep_plant.c) for the
# simulated motor and generates sstep_plant_params.h from its parameter files:
#
#   params/${CONFIG_SSTEP_PLANT_MOTOR_PARAMS}.json  motor (tools/export_params.py)
#   params/${CONFIG_SSTEP_PLANT_STAGE_PARAMS}.json  power stage (sfs-stepper-power-spice)
#
# The model is the one of the Xcos plant block: SSTEP_PLANT_FLOAT=1 (Cortex-M4F
# FPU), SSTEP_PLANT_SWITCHING=0 (averaged stage only). A change of a JSON file
# re-runs CMake.

if(CONFIG_SSTEP_PLANT)

  # <topdir>/doppelstep/apps/modules/sfs-stepper-sim/zephyr -> <topdir>/modules/lib/sfs-stepper-sim
  get_filename_component(sstep_sim_default_dir
    "${CMAKE_CURRENT_LIST_DIR}/../../../../../modules/lib/sfs-stepper-sim" ABSOLUTE)
  set(SSTEP_SIM_SOURCE_DIR "${sstep_sim_default_dir}" CACHE PATH "sfs-stepper-sim source tree")

  set(plant_src "${SSTEP_SIM_SOURCE_DIR}/sil/c/sstep_plant.c")
  set(motor_json "${SSTEP_SIM_SOURCE_DIR}/params/${CONFIG_SSTEP_PLANT_MOTOR_PARAMS}.json")
  set(stage_json "${SSTEP_SIM_SOURCE_DIR}/params/${CONFIG_SSTEP_PLANT_STAGE_PARAMS}.json")
  foreach(file "${plant_src}" "${motor_json}" "${stage_json}")
    if(NOT EXISTS "${file}")
      message(FATAL_ERROR
        "${file} not found. Run `west update` or set SSTEP_SIM_SOURCE_DIR "
        "(for example to a local checkout of sfs-stepper-sim).")
    endif()
  endforeach()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${motor_json}" "${stage_json}")

  # Every user of sstep_plant.h must see the same real type
  zephyr_compile_definitions(SSTEP_PLANT_FLOAT=1 SSTEP_PLANT_SWITCHING=0)
  zephyr_include_directories("${SSTEP_SIM_SOURCE_DIR}/sil/c")

  zephyr_library_named(sstep_plant)
  zephyr_library_sources("${plant_src}")

  # --- parameters -> sstep_plant_params.h -------------------------------------
  file(READ "${motor_json}" motor)
  file(READ "${stage_json}" stage)
  foreach(doc motor stage)
    string(JSON format ERROR_VARIABLE err GET "${${doc}}" format)
    if(err OR NOT format EQUAL 1)
      message(FATAL_ERROR "${${doc}_json}: format ${format} (expected 1); update this glue")
    endif()
  endforeach()

  set(lines "")
  foreach(key R L Kt Ns J B Td Tc w_c)
    string(JSON value GET "${motor}" ${key})
    string(TOUPPER "${key}" macro)
    string(APPEND lines "#define SSTEP_PLANT_PARAM_${macro} ${value}\n")
  endforeach()
  string(JSON motor_name GET "${motor}" name)
  foreach(key vbus fs r_leg vd dead_eff_ns)
    string(JSON value GET "${stage}" ${key})
    string(TOUPPER "${key}" macro)
    string(APPEND lines "#define SSTEP_PLANT_PARAM_STAGE_${macro} ${value}\n")
  endforeach()

  string(JSON n LENGTH "${stage}" leg_error_table)
  set(err_i "")
  set(err_v "")
  if(n GREATER 0)
    math(EXPR last "${n} - 1")
    foreach(k RANGE ${last})
      string(JSON i GET "${stage}" leg_error_table ${k} 0)
      string(JSON v GET "${stage}" leg_error_table ${k} 1)
      list(APPEND err_i "${i}")
      list(APPEND err_v "${v}")
    endforeach()
  endif()
  list(JOIN err_i ", " err_i)
  list(JOIN err_v ", " err_v)

  set(header "${CMAKE_BINARY_DIR}/zephyr/include/generated/sstep_plant_params.h")
  file(CONFIGURE OUTPUT "${header}" @ONLY CONTENT
"/* Generated by apps/modules/sfs-stepper-sim/zephyr/CMakeLists.txt: do not edit.
 * Motor: ${motor_json} (@motor_name@)
 * Stage: ${stage_json}
 * SI units (see the \"units\" of the motor file); leg error table: A, V.
 */
#ifndef SSTEP_PLANT_PARAMS_H
#define SSTEP_PLANT_PARAMS_H

#define SSTEP_PLANT_PARAM_MOTOR_NAME \"@motor_name@\"
@lines@
#define SSTEP_PLANT_PARAM_LEG_ERROR_N ${n}
#define SSTEP_PLANT_PARAM_LEG_ERROR_I {${err_i}}
#define SSTEP_PLANT_PARAM_LEG_ERROR_V {${err_v}}

#endif /* SSTEP_PLANT_PARAMS_H */
")

  message(STATUS "sfs-stepper-sim plant model: ${SSTEP_SIM_SOURCE_DIR}, "
                 "${CONFIG_SSTEP_PLANT_MOTOR_PARAMS} + ${CONFIG_SSTEP_PLANT_STAGE_PARAMS}")

endif()
