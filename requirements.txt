# Host-side Python tools of Doppelstep: the GUI (tools/gui) and the
# documentation with its PDF manual (mkdocs.yml, tools/docs). The firmware does
# not need them; its toolchain lives in the Docker image (Dockerfile).
#
# The VS Code tasks create the environment on first use:
#   Windows host:              .venv        (.vscode/host-menu.ps1)
#   Linux / Dev Container:     .venv, or .venv-linux when .venv is a Windows
#                              environment in the bind-mounted folder
#                              (.vscode/workspace-menu.sh)
#
# By hand:
#   Windows (PowerShell):  py -3 -m venv .venv
#                          .\.venv\Scripts\python.exe -m pip install -r requirements.txt
#   Linux:                 python3 -m venv .venv
#                          .venv/bin/python -m pip install -r requirements.txt
#
# Tkinter (GUI) is part of Python and cannot be installed with pip: it is in
# the python.org Windows installer ("tcl/tk and IDLE"); on Debian/Ubuntu install
# python3-tk and python3-venv with apt.

# tools/gui - serial link to the DYNAMIXEL bus (Protocol 2.0) and live plots.
# Python 3.11 or newer (smpmgr, matplotlib).
pyserial==3.5
matplotlib==3.11.2

# Firmware update over USART2: SMP client for MCUboot serial recovery
# (docs/firmware-update.md). The GUI uses its library smpclient (7.3.0).
smpmgr==0.19.1

# Documentation (HTML site and PDF manual). MkDocs 2.0 removes the plugin
# system used by the theme.
mkdocs>=1.6,<2
mkdocs-material>=9.5,<10
# PDF manual (tools/docs/build_manual.py). WeasyPrint needs the Pango library:
# Linux: apt install libpango-1.0-0 libpangoft2-1.0-0; Windows: MSYS2
# mingw-w64-x86_64-pango (found in C:\msys64\mingw64\bin automatically).
weasyprint>=62
